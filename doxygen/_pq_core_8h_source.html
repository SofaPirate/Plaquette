656// Catch operations such as: flowable >> nonFlowable;

658struct flow_error {
659 static_assert(always_false<T>::value, PQ_FLOW_OPERATOR_ERROR);

824bool Engine::timeStep() {
825 // Calculate true sample rate.
826 _updateGlobalMicroSeconds();
827
828 // Compute inter-step time.
829 _deltaTimeMicroSeconds = _totalGlobalMicroSeconds.micros32.base - _microSeconds.micros32.base;
830 float trueSampleRate = (_deltaTimeMicroSeconds ? SECONDS_TO_MICROS / _deltaTimeMicroSeconds : PLAQUETTE_MAX_SAMPLE_RATE);
831
832 // If autoSampleRate is off: wait in order to synchronize seconds with real time.
833 if (!_autoSampleRate) {
834
835 // Initilize step state.
836 if (_stepState == STEP_INIT) {
837 // Target time = current time + 1/_targetSampleRate
838 _targetTime = _microSeconds;
839 _targetTime.micros32.base += _targetDeltaTimeMicroSeconds;
840
841 // Check for overflow.
842 if (_targetTime.micros32.base >= _microSeconds.micros32.base) { // if target time is in the future: no overflow.
843 _stepState = STEP_WAIT;
844 }
845 else { // overflow
846 _targetTime.micros32.overflows++;
847 _stepState = STEP_WAIT_OVERFLOW;
848 }
849 }
850
851 // Process waiting state: will return false if still waiting, otherwise will complete the function.
852 if (_stepState == STEP_WAIT) {
853 // Still needs to wait.
854 if (_totalGlobalMicroSeconds.micros32.base < _targetTime.micros32.base)
855 return false; // break
856 }
857 else { // stepState == STEP_WAIT_OVERFLOW
858 // Still needs to wait.
859 if (_totalGlobalMicroSeconds.micros64 < _targetTime.micros64)
860 return false; // break
861 }
862
863 // Reset state.
864 _stepState = STEP_INIT;
865 }
866
867 // Update sample rate and current time to "true" / actual values.
868 _setSampleRate(trueSampleRate);
869
870 // Calculate delta time in fixed point.
871 uint64_t deltaTimeMicroSeconds64 = (uint64_t)_deltaTimeMicroSeconds;
872 _deltaTimeSecondsTimesFixed32Max = ((deltaTimeMicroSeconds64 << 32) - deltaTimeMicroSeconds64) * MICROS_TO_SECONDS;
873
874 // Sync reference time with global (true) time.
875 _microSeconds = _totalGlobalMicroSeconds;
876
877 // Increment step.
878 _nSteps++;
879
880 return true;
881}

887bool Engine::step() {
888 // On first run: do a post-begin.
889 if (_firstRun) { // the compiler should make this branching step effectively CPU-free
890 postBegin();
891 _firstRun = false;
892 return false;
893 }
894
895 // Otherwise: do a step.
896 else if (timeStep()) { // timeStep() will return false if we need to wait due to restrictive sampleRate(float)
897 // Do the pre-step.
898 preStep();
899 return true;
900 }
901 else
902 return false;
903}
